A loop, and the IR it lowers to

    fn sum_to(n: i64) -> i64 {
        var s: i64 = 0;
        var i: i64 = 1;
        while i <= n {
            s = s + i;
            i = i + 1;
        }
        return s;
    }

    fn main() {
        println(int_to_str(sum_to(100)));
    }
